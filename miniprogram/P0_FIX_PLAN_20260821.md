## P0-5 三个测验页进入即崩（引擎缺 API）

**位置**：`miniprogram/packageLearn/utils/quiz-engine.js`（只导出 generateCardsQuiz）；调用方：cards / poems / english 三个 quiz 页 + quiz-card 组件共 16 处。

**修复**：在 quiz-engine.js 中补齐四个 API 并导出（复用现有生成逻辑）：

```js
// 按模块取数据源；poems/english 的加载函数名以 learn-data.js 实际导出为准
function loadSourceByModule(module) {
  if (module === 'poems') return learnData.loadAllPoems()
  if (module === 'english') return learnData.loadAllEnglishWords()
  return learnData.loadAllCards()
}

// 从 generateCardsQuiz 内层抽出"单题生成"，cards/poems/english 共用
function buildQuestion(source, card, quizType) { /* 原 switch 逻辑原样搬入 */ }

function createQuiz(module, quizType, count) {
  var source = loadSourceByModule(module)
  var questions = []
  for (var i = 0; i < count; i++) {
    var card = source[Math.floor(Math.random() * source.length)]
    questions.push(buildQuestion(source, card, quizType))
  }
  return { module: module, quizType: quizType, questions: questions, currentIndex: 0, correctCount: 0 }
}

function submitAnswer(quiz, selectedOption) {
  var q = quiz.questions[quiz.currentIndex]
  var isCorrect = q.options[selectedOption] === q.answer
  if (isCorrect) quiz.correctCount++
  return { isCorrect: isCorrect, answer: q.answer }
}

function nextQuestion(quiz) {
  if (quiz.currentIndex + 1 >= quiz.questions.length) return false
  quiz.currentIndex++
  return true
}

function getQuizResult(quiz) {
  var total = quiz.questions.length
  var correct = quiz.correctCount || 0
  return { total: total, correct: correct, accuracy: total ? Math.round(correct / total * 100) : 0 }
}

module.exports = {
  generateCardsQuiz: generateCardsQuiz,
  createQuiz: createQuiz,
  submitAnswer: submitAnswer,
  nextQuestion: nextQuestion,
  getQuizResult: getQuizResult
}
```

**备选止血**：若测验功能暂不投入人力，可将 cards/poems/english 三个 quiz 页面从 `app.json` 移除注册并下线 quiz-card 入口——崩溃页不可达即止血。二选一，推荐补齐（功能本身有价值且页面已可分享直达）。

**验证**：三个测验页各完整答一轮 10 题：选题→答题判定→下一题→结算页正确率无误；quiz-card 组件内同样流程通过。

## P0-8 多家庭串数据

**位置**：`record/index.js:45-50`、`interaction/index.js:35-40`、`ai-chat/index.js:205-210`（三处同型的 getMemberByOpenid）
**现状**（已核实 record 处）：

```js
async function getMemberByOpenid(openid) {
  const res = await db.collection('familyMembers')
    .where({ openid, status: 'active' }).get()
  return res.data[0] || null      // ← 固定取第一条
}
```

**修复**（三步）：

1. 云函数签名加 familyId，查询条件化：

```js
async function getMemberByOpenid(openid, familyId) {
  const cond = familyId
    ? { openid, familyId, status: 'active' }
    : { openid, status: 'active' }          // 兼容未升级的旧客户端
  const res = await db.collection('familyMembers').where(cond).get()
  return res.data[0] || null
}
```

2. 三个云函数的全部 action 入口透传 `event.familyId` 给 getMemberByOpenid（grep 各文件调用点逐一替换）；
3. 客户端所有相关 `wx.cloud.callFunction` 的 data 里带上当前家庭 id（来源：app 全局态的家庭切换字段），切换家庭时同步更新。
